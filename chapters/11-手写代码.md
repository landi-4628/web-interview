## 1. 防抖 debounce

```javascript
function debounce(fn, delay = 300) {
  let timer = null;
  return function (...args) {
    clearTimeout(timer);
    timer = setTimeout(() => fn.apply(this, args), delay);
  };
}
```

## 2. 节流 throttle

```javascript
function throttle(fn, interval = 300) {
  let last = 0;
  return function (...args) {
    const now = Date.now();
    if (now - last >= interval) {
      last = now;
      fn.apply(this, args);
    }
  };
}
```

## 3. 深拷贝（简化版，面试够用）

```javascript
function deepClone(target, map = new WeakMap()) {
  if (target === null || typeof target !== 'object') return target;
  if (map.has(target)) return map.get(target); // 循环引用

  if (target instanceof Date) return new Date(target);
  if (target instanceof RegExp) return new RegExp(target);

  const clone = Array.isArray(target) ? [] : {};
  map.set(target, clone);

  for (const key of Reflect.ownKeys(target)) {
    clone[key] = deepClone(target[key], map);
  }
  return clone;
}
```

## 4. 简易 Promise.all

```javascript
function promiseAll(list) {
  return new Promise((resolve, reject) => {
    const result = [];
    let count = 0;
    if (list.length === 0) return resolve([]);

    list.forEach((p, i) => {
      Promise.resolve(p).then(
        (val) => {
          result[i] = val;
          if (++count === list.length) resolve(result);
        },
        reject
      );
    });
  });
}
```

## 5. call

```javascript
Function.prototype.myCall = function (ctx, ...args) {
  ctx = ctx ?? globalThis;
  const key = Symbol('fn');
  ctx[key] = this;
  const res = ctx[key](...args);
  delete ctx[key];
  return res;
};
```

## 6. bind

```javascript
Function.prototype.myBind = function (ctx, ...preset) {
  const fn = this;
  return function (...args) {
    return fn.apply(ctx, [...preset, ...args]);
  };
};
```

## 7. 数组扁平化

```javascript
function flatten(arr, depth = 1) {
  return depth > 0
    ? arr.reduce(
        (acc, cur) =>
          acc.concat(Array.isArray(cur) ? flatten(cur, depth - 1) : cur),
        []
      )
    : arr.slice();
}
flatten([1, [2, [3, [4]]]], 2); // [1, 2, 3, [4]]
```

## 8. EventBus

```javascript
class EventBus {
  #map = new Map();
  on(type, fn) {
    if (!this.#map.has(type)) this.#map.set(type, new Set());
    this.#map.get(type).add(fn);
  }
  off(type, fn) {
    this.#map.get(type)?.delete(fn);
  }
  emit(type, payload) {
    this.#map.get(type)?.forEach((fn) => fn(payload));
  }
}
```
