## 1. 响应式原理（defineProperty）

```javascript
// 简化版：Vue2 如何「劫持」数据
function defineReactive(obj, key, val) {
  const dep = []; // 依赖收集器
  Object.defineProperty(obj, key, {
    get() {
      // 模板编译时 Watcher 会读这里 → 把 Watcher 推进 dep
      if (Watcher.active) dep.push(Watcher.active);
      return val;
    },
    set(newVal) {
      if (newVal === val) return;
      val = newVal;
      dep.forEach((w) => w.update()); // 通知更新视图
    },
  });
}
```

**局限案例**：

```javascript
const vm = { user: { name: 'a' } };
// 下面两种 Vue2 默认监听不到，要用 Vue.set
vm.user.age = 18;
vm.items[0] = 'new';
```

## 3. v-for 与 v-if 不要写在一起

```vue
<!-- ❌ Vue2：先循环 1000 次，再判断 v-if -->
<li v-for="item in list" v-if="item.show" :key="item.id">

<!-- ✅ 先用计算属性过滤 -->
<li v-for="item in visibleList" :key="item.id">
```

```javascript
computed: {
  visibleList() {
    return this.list.filter((i) => i.show);
  },
},
```

## 4. key 为什么不能用 index

```javascript
// 列表 [A, B, C]，在头插入 X → [X, A, B, C]
// index 全变：0→A 变成 0→X，Vue 可能复用错 DOM，输入框内容串行
// 用唯一 id：Vue 知道只是「移动/新增」节点
```

## 5. computed vs watch

```javascript
export default {
  data: () => ({ list: [], keyword: '' }),
  computed: {
    // 有缓存：keyword、list 不变不会重算
    filtered() {
      return this.list.filter((i) => i.name.includes(this.keyword));
    },
  },
  watch: {
    // 适合异步
    keyword(val) {
      this.fetchSuggest(val);
    },
  },
};
```

## 8. 路由守卫（导航守卫）

**场景**：权限控制、登录验证、页面 title 切换。

### 三类守卫对比

| 类型 | 触发时机 | 常用场景 |
| --- | --- | --- |
| `beforeEach` | 路由切换**前** | 登录验证、全局权限 |
| `beforeEnter` | 路由配置**内** | 单路由权限控制 |
| `beforeRouteEnter` | 组件**实例前** | 获取组件实例（用 next） |
| `beforeRouteUpdate` | 路由参数**变时** | 详情页刷新数据 |
| `afterEach` | 路由切换**后** | 页面 title、滚动位置 |

### 全局守卫 beforeEach（最常用）

```javascript
// router/index.js
import router from './router';
import store from './store';
import NProgress from 'nprogress';

router.beforeEach(async (to, from, next) => {
  // 1. 开启进度条
  NProgress.start();

  // 2. 设置页面 title
  document.title = to.meta.title || '默认标题';

  // 3. 判断是否需要登录
  const needLogin = to.meta.requiresAuth;
  const isLoggedIn = store.state.user.token;

  if (needLogin && !isLoggedIn) {
    // 没有 token，跳转登录页，并记录要去的页面
    next({ path: '/login', query: { redirect: to.fullPath } });
  } else {
    next(); // 放行
  }
});

router.afterEach(() => {
  // 4. 关闭进度条
  NProgress.done();
  // 5. 回到顶部
  window.scrollTo(0, 0);
});
```

### 路由独享守卫 beforeEnter

```javascript
// 路由配置中单独设置
{
  path: '/admin',
  component: Admin,
  beforeEnter: (to, from, next) => {
    // 只有管理员能进
    if (store.state.user.role === 'admin') {
      next();
    } else {
      next('/403');
    }
  },
}
```

### 组件内守卫

```javascript
// UserDetail.vue
export default {
  name: 'UserDetail',
  beforeRouteEnter(to, from, next) {
    // 此时组件实例还没创建，next 里能拿到 vm
    next((vm) => {
      vm.fetchUser(to.params.id);
    });
  },
  beforeRouteUpdate(to, from, next) {
    // 路由参数变了（如 /user/1 → /user/2），复用组件
    this.fetchUser(to.params.id);
    next();
  },
  beforeRouteLeave(to, from, next) {
    // 离开时提示用户未保存的内容
    if (this.hasUnsaved) {
      const ok = confirm('有未保存的内容，确定离开？');
      ok ? next() : next(false);
    } else {
      next();
    }
  },
};
```

### 完整权限流程图

```
用户点击路由
    ↓
beforeEach 全局守卫
    ↓ 检查 token
├── 没 token + 需要登录 → 跳转 /login?redirect=xxx
├── 有 token → next()
└── 路由配置 beforeEnter（可选）
    ↓
组件 beforeRouteEnter
    ↓ next(vm => vm.xxx())
组件渲染
    ↓
afterEach 全局守卫
    ↓ 设置 title、进度条结束
```
